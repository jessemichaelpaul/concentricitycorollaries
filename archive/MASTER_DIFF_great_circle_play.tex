\begin{lemma}[The shared circle]\label{lem:great-circle}
\uses{rmk:compactify}
For $I \ne \pm J \in S^6$: $\CC_I^* \cap \CC_J^* = \RR \cup \{N\}$. The
compactified real axis is the unique circle through $N$ common to every slice
sphere, and it is precisely the fixed locus of $\Gtwo$ acting on $\OO^*$.
\end{lemma}

\begin{lemma}[The normalization fan]\label{lem:fan}
\uses{lem:great-circle, lem:residue-spheres}
On $\OO \setminus \RR$, the normalization $q = \sigma + I(q)\,\tau$ with
$I(q) = \operatorname{Im} q / |\operatorname{Im} q| \in S^6$, $\tau > 0$,
exhibits each set $\{\sigma + I\tau : I \in S^6\}$ as a single transitive
$\Gtwo$-orbit $6$-sphere. The fan is defined off the shared circle only: the
direction $I(q)$ admits no continuous extension to $\RR$
\textup{(\cite[Rem.~2.1]{GPVwind})}.
\end{lemma}

\begin{lemma}[Level invariance of the assembling symmetries]\label{lem:level-invariance}
\uses{lem:fan, def:base}
Every arrow of the assembled diagram fixes the level. $\Gtwo$ --- hence every
$U(1) \le \Gtwo$ --- fixes $\RR$ pointwise and preserves $|\operatorname{Im}|$,
so it preserves $(\sigma, \tau)$ on the nose. The band $U(1)$ fixes $0$,
$\infty$, and the modulus, so it preserves the level of the degenerate fibre
\textup{(}Lemma~\ref{lem:exp-degenerate}\textup{)}. Consequently the level is
a conserved quantity along every zigzag of $\mathcal T_A$.
\end{lemma}

\begin{proposition}[The placement sentence, great-circle register — candidate
for the \TODO{} slot]\label{prop:placement-gc}
\uses{lem:great-circle, lem:fan, lem:level-invariance, lem:exp-degenerate,

The degenerate fibre of the unique tame transport attached to an $A$-section

\paragraph{The residual goal, displayed.}
After Lemmas \ref{lem:great-circle}--\ref{lem:level-invariance} discharge the
assembly, the winding closure \textup{(C1, Cor.~5.13)}, and the per-fibre
structure \textup{(}Lemma~\ref{lem:exp-degenerate}\textup{)}, the obligation
of Proposition~\ref{prop:placement-gc} is exactly:
\[
  \forall\, n\, m:\ \operatorname{level}(q_n) \;=\; \operatorname{level}(q_m).
\]
Three formal observations, each tagged, none a verdict:
\begin{itemize}
\item Every arrow supplied by the chain conserves the quantity the goal
  equates \textup{(}Lemma~\ref{lem:level-invariance}; repo:
  \texttt{zigzag\_iff\_level}\textup{)}. A closing term therefore consumes
  input not among the arrows of the chain.
\item \textup{C4} makes the index set infinite. The goal is an equality
  statement quantified over that set; enlarging the set enlarges the
  quantifier's range and leaves the equality obligation unchanged, because
  Lemma~\ref{lem:level-invariance} is uniform in $n$. Infinitude is consumed
  by the fibre's infinitude \textup{(}the $k$-ladder\textup{)}, not by the
  placement.
\item The fan cannot carry the goal through the circle: the normalization has
  no continuous extension to $\RR$ \textup{(\cite[Rem.~2.1]{GPVwind},
  the author's pin)}; the gluing of slices is along $\RR \cup \{N\}$
  pointwise, and $\nfr$ is the pole \textup{(C1)}, not a zero.
\end{itemize}
